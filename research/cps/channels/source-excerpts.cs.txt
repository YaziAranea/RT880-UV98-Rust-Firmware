// RT880 MainForm.cs:10779; CopyArrayToString
	public byte[] CopyArrayToString(byte[] a, int offset, int lenth)
	{
		byte[] array = new byte[lenth];
		for (int i = 0; i < lenth; i++)
		{
			if (a[offset + i] == byte.MaxValue)
			{
				return array;
			}
			array[i] = a[offset + i];
		}
		return array;
	}

	public void SaveCHName(int ch)
	{
		byte[] bytes = utf8.GetBytes(listChannel.Items[ch].SubItems[15].Text);
		bytes = Encoding.Convert(utf8, gbk, bytes);
		if (bytes.Length <= 16)
		{
			for (int i = 0; i < bytes.Length; i++)
			{
				bufAll[ch * 48 + 32 + i] = bytes[i];
			}
			for (int i = bytes.Length; i < 16; i++)
			{
				bufAll[ch * 48 + 32 + i] = byte.MaxValue;
			}
		}
		else
		{
			for (int i = 0; i < 16; i++)
			{
				bufAll[ch * 48 + 32 + i] = bytes[i];
			}
		}
	}

// RT880 MainForm.cs:11493; CheckFreq
	public string CheckFreq(string s)
	{
		int num = 0;
		int length = 0;
		string text = "";
		for (int i = 0; i < s.Length; i++)
		{
			if (s[i] == '.')
			{
				num++;
				length = i;
			}
			else if (s[i] < '0' || s[i] > '9')
			{
				return "";
			}
		}
		if (num > 1)
		{
			return "";
		}
		int num2;
		if (num == 1)
		{
			text = s.Substring(0, length);
			num2 = Convert.ToInt32(text);
			if (num2 < 18 || (num2 >= 660 && num2 < 840) || num2 >= 1000)
			{
				return "";
			}
			text = s;
			if (num2 < 100)
			{
				if (text.Length == 9)
				{
					text = text.Substring(0, 8);
				}
				while (text.Length < 8)
				{
					text += "0";
				}
			}
			else
			{
				while (text.Length < 9)
				{
					text += "0";
				}
			}
			return text;
		}
		num2 = Convert.ToInt32(s);
		if (num2 < 18 || (num2 >= 660 && num2 < 840) || num2 >= 1000)
		{
			return "";
		}
		text = s + ".";
		if (num2 < 100)
		{
			while (text.Length < 8)
			{
				text += "0";
			}
		}
		else
		{
			while (text.Length < 9)
			{
				text += "0";
			}
		}
		return text;
	}


// RT880 MainForm.cs:11729; SaveSubaudio
	public int SaveSubaudio(string Subaudio, bool encodeflag)
	{
		int num = 0;
		try
		{
			if (Subaudio[0] != 'D')
			{
				num = Convert.ToInt32(Subaudio.Replace(".", ""));
				num &= 0xFFF;
				return num | 0x1000;
			}
			num = ((byte)Subaudio[1] - 48) * 64 + ((byte)Subaudio[2] - 48) * 8 + ((byte)Subaudio[3] - 48);
			num &= 0xFFF;
			if (Subaudio[4] == 'I')
			{
				return num | 0x3000;
			}
			return num | 0x2000;
		}
		catch (Exception)
		{
			return 0;
		}
	}


// RT880 MainForm.cs:11754; SetSubaudio
	public string SetSubaudio(byte[] Subaudio)
	{
		string text = "";
		int num = 0;
		num = Subaudio[0] & 0xF;
		num <<= 8;
		num += Subaudio[1];
		if ((Subaudio[0] & 0xF0) == 16)
		{
			return num / 10 + "." + num % 10;
		}
		if ((Subaudio[0] & 0xF0) == 32)
		{
			return DCS_NO[num] + "N";
		}
		if ((Subaudio[0] & 0x30) == 48)
		{
			return DCS_NO[num] + "I";
		}
		if (flgLanguage == 0)
		{
			return "无";
		}
		if (flgLanguage == 0)
		{
			return "無";
		}
		return "None";
	}
